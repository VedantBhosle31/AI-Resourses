\section{Chapter 2: Classification Models \& Architectures (30
Models)}\label{chapter-2-classification-models-architectures-30-models}

\subsection{1. LinearSVC}\label{linearsvc}

\subsubsection{Mental Model}\label{mental-model}

Find a linear hyperplane that separates classes with the largest
possible geometric margin.

\subsubsection{Mathematical Formulation \&
Prediction}\label{mathematical-formulation-prediction}

\textbf{f(x) = w\^{}T * x + b\\
\strut \\
Prediction: y\_hat = +1 if f(x) \textgreater{} 0 else -1}

\subsubsection{Optimization \& Loss
Function}\label{optimization-loss-function}

min\_\{w,b\} 0.5 * \textbar\textbar w\textbar\textbar² + C * Σ max(0, 1
- y\_i * (w\^{}T * x\_i + b))²\\
\strut \\
The loss term is Squared Hinge Loss (or standard Hinge Loss).

\subsubsection{Meaning Breakdown \& Mental
Flowchart}\label{meaning-breakdown-mental-flowchart}

\begin{quote}
\textbf{\textbar\textbar w\textbar\textbar²\\
↓\\
}Keep boundary simple / maximize geometric margin

\textbf{hinge loss\\
↓\\
}Punish incorrectly classified or insufficiently separated samples

\textbf{C hyperparameter\\
↓\\
}Trade-off between wide margin and classification error
\end{quote}

\subsection{2. SGDClassifier}\label{sgdclassifier}

\subsubsection{Mental Model}\label{mental-model-1}

Take a small sample or single data point, calculate the prediction
error, and instantly update weights via gradient step.

\subsubsection{Mathematical Formulation \&
Prediction}\label{mathematical-formulation-prediction-1}

\textbf{General Objective: J(w) = (1/n) * Σ L(y\_i, f(x\_i)) + λ *
R(w)\\
\strut \\
Single Step Update: w\_\{t+1\} = w\_t - η * ∇J(w\_t)}

\subsubsection{Optimization \& Loss
Function}\label{optimization-loss-function-1}

Configurable loss L(y, f(x)):\\
- L = Hinge (SVM)\\
- L = Log loss (Logistic Regression)\\
- L = Modified Huber (probabilistic SVM)\\
\strut \\
Regularization R(w): L2 norm 0.5*\textbar\textbar w\textbar\textbar², L1
norm \textbar\textbar w\textbar\textbar\_1, or ElasticNet.

\subsubsection{Meaning Breakdown \& Mental
Flowchart}\label{meaning-breakdown-mental-flowchart-1}

\begin{quote}
\textbf{prediction\\
↓\\
}Make quick output guess for current sample

\textbf{calculate error\\
↓\\
}Compute loss value L(y\_i, y\_hat\_i)

\textbf{calculate gradient\\
↓\\
}Compute gradient ∇J with respect to weight vector w

\textbf{change weights\\
↓\\
}Update weights w\_\{t+1\} = w\_t - η * ∇J and repeat
\end{quote}

\subsection{3. MLPClassifier}\label{mlpclassifier}

\subsubsection{Mental Model}\label{mental-model-2}

Stack multiple layers of artificial neurons to learn complex, non-linear
feature representations automatically.

\subsubsection{Mathematical Formulation \&
Prediction}\label{mathematical-formulation-prediction-2}

\textbf{Single Neuron: z = w\^{}T * x + b, Activation: a = f(z)\\
\strut \\
Network Layers: h\_1 = f(W\_1 * x + b\_1), h\_2 = f(W\_2 * h\_1 + b\_2),
y\_hat = Softmax(W\_3 * h\_2 + b\_3)\\
\strut \\
Softmax Probability: P(y=k\textbar x) = exp(z\_k) / Σ\_j exp(z\_j)}

\subsubsection{Optimization \& Loss
Function}\label{optimization-loss-function-2}

Cross-Entropy Loss: L = - Σ y\_k * log(y\_hat\_k) + (α / 2n) * Σ
\textbar\textbar W\_l\textbar\textbar\_F²\\
\strut \\
Training uses Backpropagation + Gradient Descent (Adam, L-BFGS, or SGD).

\subsubsection{Meaning Breakdown \& Mental
Flowchart}\label{meaning-breakdown-mental-flowchart-2}

\begin{quote}
\textbf{input layer\\
↓\\
}Receive raw input feature vector x

\textbf{hidden layers\\
↓\\
}Extract non-linear hierarchical feature embeddings

\textbf{softmax output\\
↓\\
}Produce normalized probability distribution across classes

\textbf{backprop\\
↓\\
}Propagate gradient error backwards to update weight matrices W\_l
\end{quote}

\subsection{4. Perceptron}\label{perceptron}

\subsubsection{Mental Model}\label{mental-model-3}

A single linear neuron that tests predictions and shifts its boundary
weights only when it makes a mistake.

\subsubsection{Mathematical Formulation \&
Prediction}\label{mathematical-formulation-prediction-3}

\textbf{Prediction: y\_hat = sgn(w\^{}T * x + b)\\
\strut \\
Condition to update: If y\_i * (w\^{}T * x\_i + b) ≤ 0 (Misclassified)}

\subsubsection{Optimization \& Loss
Function}\label{optimization-loss-function-3}

Update Rule:\\
w ← w + η * y\_i * x\_i\\
b ← b + η * y\_i

\subsubsection{Meaning Breakdown \& Mental
Flowchart}\label{meaning-breakdown-mental-flowchart-3}

\begin{quote}
\textbf{correct prediction\\
↓\\
}Do nothing (weights stay unchanged)

\textbf{misclassified sample\\
↓\\
}Shift weight vector w in the direction of the error y\_i * x\_i

\textbf{simple loop\\
↓\\
}Iterate until linear convergence or max iterations reached
\end{quote}

\subsection{5. LogisticRegression}\label{logisticregression}

\subsubsection{Mental Model}\label{mental-model-4}

Compute a linear score for a sample, then map it through a Sigmoid curve
to output a clean class probability between 0 and 1.

\subsubsection{Mathematical Formulation \&
Prediction}\label{mathematical-formulation-prediction-4}

\textbf{Linear Score: z = w\^{}T * x + b\\
\strut \\
Sigmoid Function: P(y=1\textbar x) = σ(z) = 1 / (1 + exp(-z))}

\subsubsection{Optimization \& Loss
Function}\label{optimization-loss-function-4}

Binary Cross-Entropy Loss:\\
L(w) = - (1/n) * Σ {[} y\_i * log(p\_i) + (1 - y\_i) * log(1 - p\_i) {]}
+ (1 / 2C) * \textbar\textbar w\textbar\textbar²\\
\strut \\
Optimized via L-BFGS, Newton-CG, or Liblinear.

\subsubsection{Meaning Breakdown \& Mental
Flowchart}\label{meaning-breakdown-mental-flowchart-4}

\begin{quote}
\textbf{linear score z\\
↓\\
}Combine features linearly z = w\^{}T * x + b

\textbf{sigmoid σ(z)\\
↓\\
}Squash z to probability scale {[}0, 1{]}

\textbf{cross-entropy\\
↓\\
}Penalize confident wrong probability guesses exponentially

\textbf{gradient descent\\
↓\\
}Adjust w to maximize log-likelihood of dataset
\end{quote}

\subsection{6. LogisticRegressionCV}\label{logisticregressioncv}

\subsubsection{Mental Model}\label{mental-model-5}

Fit Logistic Regression across multiple internal cross-validation folds
to automatically discover the best regularization strength C.

\subsubsection{Mathematical Formulation \&
Prediction}\label{mathematical-formulation-prediction-5}

\textbf{P(y=1\textbar x) = σ(w\^{}T * x + b)\\
\strut \\
Validation Score: CV\_Score(C\_k) = (1/K) * Σ\_\{v=1\}\^{}K
Accuracy\_val( C\_k, Fold\_v )}

\subsubsection{Optimization \& Loss
Function}\label{optimization-loss-function-5}

Solves internal grid search over a grid of C values:\\
C\^{}* = argmax\_\{C\_k\} CV\_Score(C\_k)\\
\strut \\
Final fit uses C\^{}* on the full dataset.

\subsubsection{Meaning Breakdown \& Mental
Flowchart}\label{meaning-breakdown-mental-flowchart-5}

\begin{quote}
\textbf{C grid search\\
↓\\
}Test multiple regularization candidate values C

\textbf{K-fold split\\
↓\\
}Evaluate out-of-fold cross-validation accuracy for each C

\textbf{select optimal C*\\
↓\\
}Pick C* with maximum validation score

\textbf{final model\\
↓\\
}Fit Logistic Regression using optimal C*
\end{quote}

\subsection{7. SVC (Support Vector
Classifier)}\label{svc-support-vector-classifier}

\subsubsection{Mental Model}\label{mental-model-6}

Project data into higher-dimensional space using kernel functions where
classes become linearly separable by a maximum-margin hyperplane.

\subsubsection{Mathematical Formulation \&
Prediction}\label{mathematical-formulation-prediction-6}

\textbf{Dual Decision Function: f(x) = sgn( Σ\_\{i ∈ SV\} α\_i * y\_i *
K(x\_i, x) + b )\\
\strut \\
Kernels K(x\_i, x):\\
- Linear: x\_i\^{}T * x\\
- RBF: exp(-γ * \textbar\textbar x\_i - x\textbar\textbar²)\\
- Poly: (γ * x\_i\^{}T * x + r)\^{}d}

\subsubsection{Optimization \& Loss
Function}\label{optimization-loss-function-6}

Dual Quadratic Optimization:\\
max\_α Σ α\_i - 0.5 * Σ\_i Σ\_j α\_i * α\_j * y\_i * y\_j * K(x\_i,
x\_j) s.t. 0 ≤ α\_i ≤ C, Σ α\_i * y\_i = 0\\
\strut \\
Solved using Sequential Minimal Optimization (SMO).

\subsubsection{Meaning Breakdown \& Mental
Flowchart}\label{meaning-breakdown-mental-flowchart-6}

\begin{quote}
\textbf{kernel K(x, x\_i)\\
↓\\
}Compute non-linear similarity between samples in feature space

\textbf{support vectors\\
↓\\
}Identify key boundary points where α\_i \textgreater{} 0

\textbf{max margin boundary\\
↓\\
}Construct optimal separating boundary between classes
\end{quote}

\subsection{8. CalibratedClassifierCV}\label{calibratedclassifiercv}

\subsubsection{Mental Model}\label{mental-model-7}

\subsubsection{Meaning Breakdown \& Mental
Flowchart}\label{meaning-breakdown-mental-flowchart-30}

\begin{quote}

↓\\

↓\\

\textbf{C parameter\\
↓\\
